/*
  VERY IMPORTANT:

  Your present code uses a 0-25V voltage sensor.

  DO NOT directly connect ESP32 or these cheap sensor modules
  to 230V mains electricity.

  For a college prototype using a low-voltage supply,
  keep VOLTAGE_SCALE_FACTOR = 1.0.

  If you later use a properly ISOLATED measurement circuit,
  calibrate the scaling factor experimentally.
*/

const float VOLTAGE_SENSOR_MAX = 25.0;

const float ADC_REFERENCE = 3.3;

const int ADC_MAX = 4095;

// Raw sensor voltage × this = displayed monitored voltage.
const float VOLTAGE_SCALE_FACTOR = 1.0;

/*
  Your original program had limits of 200-250 V.

  But with VOLTAGE_SENSOR_MAX = 25V and scale factor 1,
  the ESP32 can NEVER measure 200V.

  Therefore voltage-fault checking is OFF by default.

  Once your voltage measurement circuit is calibrated,
  enable it.
*/

const bool ENABLE_VOLTAGE_FAULT = false;

const float VOLTAGE_LOW = 200.0;
const float VOLTAGE_HIGH = 250.0;

void setup()
{

  );

  );

  // ----------------------------------------------------------
  // BOOT ID
  // ----------------------------------------------------------

  bootId = esp_random();

  Serial.print("Device ID : ");
  Serial.println(DEVICE_ID);

  Serial.print("Boot ID   : ");
  Serial.println(bootId);

  // ----------------------------------------------------------
  // GPIO
  // ----------------------------------------------------------

  pinMode(
    VIBRATION_PIN,
    INPUT
  );

  pinMode(
    VOLTAGE_PIN,
    INPUT
  );

  pinMode(
    BUZZER_PIN,
    OUTPUT
  );

  digitalWrite(
    BUZZER_PIN,
    LOW
  );

  // ----------------------------------------------------------
  // I2C
  // ----------------------------------------------------------

  Wire.begin(
    SDA_PIN,
    SCL_PIN
  );

  // ----------------------------------------------------------
  // LCD
  // ----------------------------------------------------------

  lcd.init();

  lcd.backlight();

  lcd.clear();

  lcd.setCursor(0, 0);

  lcd.print(
    "SMART POLE"
  );

  lcd.setCursor(0, 1);

  lcd.print(
    "STARTING..."
  );

  // ----------------------------------------------------------
  // DHT
  // ----------------------------------------------------------

  dht.begin();

  // ----------------------------------------------------------
  // MPU6050
  // ----------------------------------------------------------

  if (mpu.begin())
  {
    mpuOK = true;

    Serial.println(
      "MPU6050 OK"
    );

    mpu.setAccelerometerRange(
      MPU6050_RANGE_8_G
    );

    mpu.setGyroRange(
      MPU6050_RANGE_500_DEG
    );

    mpu.setFilterBandwidth(
      MPU6050_BAND_21_HZ
    );
  }
  else
  {
    mpuOK = false;

    Serial.println(
      "MPU6050 NOT FOUND"
    );
  }

  // ----------------------------------------------------------
  // INA219
  // ----------------------------------------------------------

  if (ina219.begin())
  {
    inaOK = true;

    Serial.println(
      "INA219 OK"
    );

    ina219.setCalibration_32V_2A();
  }
  else
  {
    inaOK = false;

    Serial.println(
      "INA219 NOT FOUND"
    );
  }

  // ----------------------------------------------------------
  // WIFI
  // ----------------------------------------------------------

  connectWiFi();

  // ----------------------------------------------------------
  // NTP TIME
  // ----------------------------------------------------------

  if (
    WiFi.status()
    ==
    WL_CONNECTED
  )
  {
    configTime(
      0,
      0,
      "pool.ntp.org",
      "time.google.com"
    );
  }

  // ----------------------------------------------------------
  // READY
  // ----------------------------------------------------------

  lcd.clear();

  lcd.setCursor(
    0,
    0
  );

  lcd.print(
    "SYSTEM READY"
  );

  lcd.setCursor(
    0,
    1
  );

  if (
    WiFi.status()
    ==
    WL_CONNECTED
  )
  {
    lcd.print(
      "WIFI CONNECTED"
    );
  }
  else
  {
    lcd.print(
      "LOCAL MODE"
    );
  }

  delay(1000);

  Serial.println(
    "System Ready"
  );
}

// ============================================================
// WIFI CONNECTION
// ============================================================

void connectWiFi()
{
  Serial.println();

  Serial.print(
    "Connecting to WiFi: "
  );

  Serial.println(
    WIFI_SSID
  );

  WiFi.mode(
    WIFI_STA
  );

  WiFi.begin(
    WIFI_SSID,
    WIFI_PASSWORD
  );

  unsigned long start =
    millis();

  while (
    WiFi.status()
      !=
    WL_CONNECTED
    &&
    millis() - start
      <
    15000
  )
  {
    delay(500);

    Serial.print(".");
  }

  Serial.println();

  if (
    WiFi.status()
    ==
    WL_CONNECTED
  )
  {
    Serial.println(
      "WiFi Connected"
    );

    Serial.print(
      "ESP32 IP: "
    );

    Serial.println(
      WiFi.localIP()
    );

    Serial.print(
      "WiFi RSSI: "
    );

    Serial.println(
      WiFi.RSSI()
    );
  }
  else
  {
    Serial.println(
      "WiFi connection failed."
    );

    Serial.println(
      "System will continue locally."
    );
  }
}

// ============================================================
// WIFI AUTO RECONNECT
// ============================================================

void maintainWiFi()
{
  if (
    WiFi.status()
    ==
    WL_CONNECTED
  )
  {
    return;
  }

  if (
    millis()
      -
    lastWiFiReconnect
      <
    WIFI_RECONNECT_INTERVAL
  )
  {
    return;
  }

  lastWiFiReconnect =
    millis();

  Serial.println(
    "Trying WiFi reconnect..."
  );

  WiFi.disconnect();

  WiFi.begin(
    WIFI_SSID,
    WIFI_PASSWORD
  );
}

// ============================================================
// READ MPU6050 VIBRATION
// ============================================================

float readVibration()
{
  if (!mpuOK)
  {
    return 0.0;
  }

  sensors_event_t accel;

  sensors_event_t gyro;

  sensors_event_t temp;

  mpu.getEvent(
    &accel,
    &gyro,
    &temp
  );

  float ax =
    accel.acceleration.x;

  float ay =
    accel.acceleration.y;

  float az =
    accel.acceleration.z;

  float totalAcceleration =
    sqrt(
      (ax * ax)
      +
      (ay * ay)
      +
      (az * az)
    );

  // Remove gravity contribution.
  float vibration =
    fabs(
      totalAcceleration
      -
      9.81
    );

  return vibration;
}

// ============================================================
// READ VOLTAGE SENSOR
// ============================================================

float readVoltageSensor()
{
  int adcValue =
    analogRead(
      VOLTAGE_PIN
    );

  float adcVoltage =
    (
      (float) adcValue
      /
      ADC_MAX
    )
    *
    ADC_REFERENCE;

  float sensorVoltage =
    (
      adcVoltage
      /
      ADC_REFERENCE
    )
    *
    VOLTAGE_SENSOR_MAX;

  float calibratedVoltage =
    sensorVoltage
    *
    VOLTAGE_SCALE_FACTOR;

  return calibratedVoltage;
}

// ============================================================
// READ INA219
// ============================================================

void readINA219()
{
  if (!inaOK)
  {
    current_mA = 0;

    current_A = 0;

    busVoltage = 0;

    return;
  }

  current_mA =
    ina219.getCurrent_mA();

  current_A =
    current_mA
    /
    1000.0;

  busVoltage =
    ina219.getBusVoltage_V();
}

// ============================================================
// READ ALL SENSORS
// ============================================================

void readSensors()
{
  // ----------------------------------------------------------
  // DHT11
  // ----------------------------------------------------------

  float newTemperature =
    dht.readTemperature();

  float newHumidity =
    dht.readHumidity();

  if (
    !isnan(newTemperature)
    &&
    !isnan(newHumidity)
  )
  {
    temperature =
      newTemperature;

    humidity =
      newHumidity;

    dhtOK = true;
  }
  else
  {
    dhtOK = false;

    Serial.println(
      "DHT SENSOR READ ERROR"
    );
  }

  // ----------------------------------------------------------
  // MPU6050
  // ----------------------------------------------------------

  vibrationValue =
    readVibration();

  // ----------------------------------------------------------
  // DIGITAL VIBRATION SENSOR
  // ----------------------------------------------------------

  vibrationDigital =
    digitalRead(
      VIBRATION_PIN
    );

  // ----------------------------------------------------------
  // INA219
  // ----------------------------------------------------------

  readINA219();

  // ----------------------------------------------------------
  // VOLTAGE SENSOR
  // ----------------------------------------------------------

  monitoredVoltage =
    readVoltageSensor();
}

// ============================================================
// CREATE SENSOR READING
// ============================================================

SensorReading createReading()
{
  SensorReading reading;

  sequenceNumber++;

  reading.sequence =
    sequenceNumber;

  reading.uptimeMs =
    millis();

  reading.temperature =
    temperature;

  reading.humidity =
    humidity;

  reading.currentA =
    current_A;

  reading.inaBusVoltage =
    busVoltage;

  reading.monitoredVoltage =
    monitoredVoltage;

  reading.vibration =
    vibrationValue;

  reading.vibrationDetected =
    (
      vibrationDigital
      ==
      VIBRATION_ACTIVE_STATE
    );

  reading.highTemperature =
    (
      temperature
      >=
      TEMP_LIMIT
    );

  reading.overCurrent =
    (
      current_A
      >=
      CURRENT_LIMIT
    );

  reading.highVibration =
    (
      vibrationValue
      >=
      VIBRATION_LIMIT
    );

  reading.underVoltage =
    false;

  reading.overVoltage =
    false;

  if (
    ENABLE_VOLTAGE_FAULT
  )
  {
    reading.underVoltage =
      (
        monitoredVoltage
        <
        VOLTAGE_LOW
      );

    reading.overVoltage =
      (
        monitoredVoltage
        >
        VOLTAGE_HIGH
      );
  }

  reading.fault =
    (
      reading.highTemperature
      ||
      reading.overCurrent
      ||
      reading.highVibration
      ||
      reading.vibrationDetected
      ||
      reading.underVoltage
      ||
      reading.overVoltage
    );

  if (
    WiFi.status()
    ==
    WL_CONNECTED
  )
  {
    reading.wifiRSSI =
      WiFi.RSSI();
  }
  else
  {
    reading.wifiRSSI =
      -999;
  }

  return reading;
}

// ============================================================
// GET INTERNET TIME
// ============================================================

unsigned long getEpochTime()
{
  time_t now;

  time(
    &now
  );

  if (
    now
    >
    1700000000
  )
  {
    return
      (unsigned long) now;
  }

  return 0;
}

// ============================================================
// HTTP/HTTPS POST
// ============================================================

bool postReading(
  const SensorReading& reading,
  const char* url
)
{
  if (
    WiFi.status()
    !=
    WL_CONNECTED
  )
  {
    return false;
  }

  // Don't try placeholder cloud URL.
  String targetURL =
    String(url);

  if (
    targetURL.indexOf(
      "YOUR-DOMAIN"
    )
    >=
    0
  )
  {
    return false;
  }

  // ----------------------------------------------------------
  // JSON
  // ----------------------------------------------------------

  StaticJsonDocument<1536> doc;

  doc["device_id"] =
    DEVICE_ID;

  doc["boot_id"] =
    bootId;

  doc["sequence"] =
    reading.sequence;

  String readingId =
    String(DEVICE_ID)
    +
    "-"
    +
    String(bootId)
    +
    "-"
    +
    String(reading.sequence);

  doc["reading_id"] =
    readingId;

  doc["uptime_ms"] =
    reading.uptimeMs;

  unsigned long epoch =
    getEpochTime();

  if (epoch > 0)
  {
    doc["timestamp_epoch"] =
      epoch;
  }
  else
  {
    doc["timestamp_epoch"] =
      nullptr;
  }

  // ----------------------------------------------------------
  // SENSOR DATA
  // ----------------------------------------------------------

  JsonObject sensors =
    doc.createNestedObject(
      "sensors"
    );

  sensors["temperature"] =
    reading.temperature;

  sensors["humidity"] =
    reading.humidity;

  sensors["current_a"] =
    reading.currentA;

  sensors["ina_bus_voltage"] =
    reading.inaBusVoltage;

  sensors["monitored_voltage"] =
    reading.monitoredVoltage;

  sensors["vibration_ms2"] =
    reading.vibration;

  sensors["vibration_detected"] =
    reading.vibrationDetected;

  // ----------------------------------------------------------
  // SENSOR HEALTH
  // ----------------------------------------------------------

  JsonObject health =
    doc.createNestedObject(
      "sensor_health"
    );

  health["dht11"] =
    dhtOK;

  health["mpu6050"] =
    mpuOK;

  health["ina219"] =
    inaOK;

  // ----------------------------------------------------------
  // FAULT INFORMATION
  // ----------------------------------------------------------

  JsonObject faults =
    doc.createNestedObject(
      "faults"
    );

  faults["high_temperature"] =
    reading.highTemperature;

  faults["over_current"] =
    reading.overCurrent;

  faults["high_vibration"] =
    reading.highVibration;

  faults["vibration_triggered"] =
    reading.vibrationDetected;

  faults["under_voltage"] =
    reading.underVoltage;

  faults["over_voltage"] =
    reading.overVoltage;

  doc["fault"] =
    reading.fault;

  // ----------------------------------------------------------
  // CONNECTION INFO
  // ----------------------------------------------------------

  JsonObject connection =
    doc.createNestedObject(
      "connection"
    );

  connection["wifi_connected"] =
    true;

  connection["wifi_rssi"] =
    reading.wifiRSSI;

  connection["ip"] =
    WiFi.localIP().toString();

  // ----------------------------------------------------------
  // THRESHOLDS
  // ----------------------------------------------------------

  JsonObject thresholds =
    doc.createNestedObject(
      "thresholds"
    );

  thresholds["temperature"] =
    TEMP_LIMIT;

  thresholds["current"] =
    CURRENT_LIMIT;

  thresholds["vibration"] =
    VIBRATION_LIMIT;

  thresholds["voltage_low"] =
    VOLTAGE_LOW;

  thresholds["voltage_high"] =
    VOLTAGE_HIGH;

  thresholds["voltage_fault_enabled"] =
    ENABLE_VOLTAGE_FAULT;

  // ----------------------------------------------------------
  // SERIALIZE
  // ----------------------------------------------------------

  String payload;

  serializeJson(
    doc,
    payload
  );

  HTTPClient http;

  http.setTimeout(
    2500
  );

  http.addHeader(
    "Content-Type",
    "application/json"
  );

  http.addHeader(
    "X-DEVICE-ID",
    DEVICE_ID
  );

  http.addHeader(
    "X-DEVICE-KEY",
    DEVICE_API_KEY
  );

  int responseCode =
    -1;

  // ----------------------------------------------------------
  // HTTPS
  // ----------------------------------------------------------

  if (
    targetURL.startsWith(
      "https://"
    )
  )
  {
    WiFiClientSecure secureClient;

    /*
      Prototype mode:

      setInsecure() means the TLS certificate is not verified.

      It encrypts traffic but doesn't authenticate the server
      certificate.

      For final production deployment we should replace this
      with the correct CA certificate.
    */

    secureClient.setInsecure();

    if (
      !http.begin(
        secureClient,
        targetURL
      )
    )
    {
      return false;
    }

    http.addHeader(
      "Content-Type",
      "application/json"
    );

    http.addHeader(
      "X-DEVICE-ID",
      DEVICE_ID
    );

    http.addHeader(
      "X-DEVICE-KEY",
      DEVICE_API_KEY
    );

    responseCode =
      http.POST(
        payload
      );
  }

  // ----------------------------------------------------------
  // HTTP
  // ----------------------------------------------------------

  else
  {
    WiFiClient client;

    if (
      !http.begin(
        client,
        targetURL
      )
    )
    {
      return false;
    }

    http.addHeader(
      "Content-Type",
      "application/json"
    );

    http.addHeader(
      "X-DEVICE-ID",
      DEVICE_ID
    );

    http.addHeader(
      "X-DEVICE-KEY",
      DEVICE_API_KEY
    );

    responseCode =
      http.POST(
        payload
      );
  }

  bool success =
    (
      responseCode
      >=
      200
      &&
      responseCode
      <
      300
    );

  if (!success)
  {
    Serial.print(
      "POST failed: "
    );

    Serial.print(
      targetURL
    );

    Serial.print(
      " -> "
    );

    Serial.println(
      responseCode
    );
  }

  http.end();

  return success;
}

// ============================================================
// ADD READING TO CLOUD BUFFER
// ============================================================

void addToCloudBuffer(
  const SensorReading& reading
)
{
  int position =
    (
      bufferHead
      +
      bufferCount
    )
    %
    BUFFER_SIZE;

  if (
    bufferCount
    <
    BUFFER_SIZE
  )
  {
    cloudBuffer[position] =
      reading;

    bufferCount++;
  }
  else
  {
    /*
      Buffer full:
      overwrite oldest reading.
    */

    cloudBuffer[bufferHead] =
      reading;

    bufferHead =
      (
        bufferHead
        +
        1
      )
      %
      BUFFER_SIZE;
  }

  Serial.print(
    "Cloud buffer: "
  );

  Serial.print(
    bufferCount
  );

  Serial.print(
    "/"
  );

  Serial.println(
    BUFFER_SIZE
  );
}

// ============================================================
// SYNC BUFFERED CLOUD DATA
// ============================================================

void syncCloudBuffer()
{
  if (
    !ENABLE_CLOUD_SERVER
    ||
    WiFi.status()
      !=
    WL_CONNECTED
    ||
    bufferCount
      <=
    0
  )
  {
    return;
  }

  // Upload at most three old readings each cycle.
  // Prevents blocking the ESP32 for too long.

  int attempts = 0;

  while (
    bufferCount
      >
    0
    &&
    attempts
      <
    3
  )
  {
    SensorReading reading =
      cloudBuffer[bufferHead];

    bool success =
      postReading(
        reading,
        CLOUD_API_URL
      );

    if (!success)
    {
      cloudServerOnline =
        false;

      return;
    }

    cloudServerOnline =
      true;

    lastCloudSuccess =
      millis();

    bufferHead =
      (
        bufferHead
        +
        1
      )
      %
      BUFFER_SIZE;

    bufferCount--;

    attempts++;

    Serial.println(
      "Buffered reading synced to cloud."
    );
  }
}

// ============================================================
// SEND CURRENT DATA
// ============================================================

void sendNetworkData()
{
  SensorReading reading =
    createReading();

  if (
    WiFi.status()
    !=
    WL_CONNECTED
  )
  {
    Serial.println(
      "No WiFi: storing reading for cloud."
    );

    addToCloudBuffer(
      reading
    );

    localServerOnline =
      false;

    cloudServerOnline =
      false;

    return;
  }

  // ----------------------------------------------------------
  // LOCAL SERVER
  // ----------------------------------------------------------

  if (
    ENABLE_LOCAL_SERVER
  )
  {
    bool localSuccess =
      postReading(
        reading,
        LOCAL_API_URL
      );

    localServerOnline =
      localSuccess;

    if (localSuccess)
    {
      lastLocalSuccess =
        millis();

      Serial.println(
        "LOCAL: data sent successfully."
      );
    }
    else
    {
      Serial.println(
        "LOCAL: server unavailable."
      );
    }
  }

  // ----------------------------------------------------------
  // CLOUD SERVER
  // ----------------------------------------------------------

  bool cloudSuccess =
    false;

  if (
    ENABLE_CLOUD_SERVER
  )
  {
    cloudSuccess =
      postReading(
        reading,
        CLOUD_API_URL
      );

    cloudServerOnline =
      cloudSuccess;

    if (
      cloudSuccess
    )
    {
      lastCloudSuccess =
        millis();

      Serial.println(
        "CLOUD: data sent successfully."
      );
    }
    else
    {
      Serial.println(
        "CLOUD: unavailable, buffering reading."
      );

      addToCloudBuffer(
        reading
      );
    }
  }

  // ----------------------------------------------------------
  // SYNC OLD CLOUD DATA
  // ----------------------------------------------------------

  if (
    cloudSuccess
  )
  {
    syncCloudBuffer();
  }
}

// ============================================================
// BUZZER
// ============================================================

void controlBuzzer(
  bool fault
)
{
  if (fault)
  {
    digitalWrite(
      BUZZER_PIN,
      HIGH
    );
  }
  else
  {
    digitalWrite(
      BUZZER_PIN,
      LOW
    );
  }
}

// ============================================================
// LCD
// ============================================================

void updateLCD()
{
  SensorReading reading =
    createReading();

  /*
    createReading() increases sequence number,
    which we don't want for LCD-only updates.

    Undo it here.
  */

  sequenceNumber--;

  lcd.clear();

  switch (
    lcdPage
  )
  {
    // --------------------------------------------------------
    // PAGE 0
    // --------------------------------------------------------

    case 0:

      lcd.setCursor(
        0,
        0
      );

      lcd.print(
        "TEMP:"
      );

      lcd.print(
        temperature,
        1
      );

      lcd.print(
        (char)223
      );

      lcd.print(
        "C"
      );

      lcd.setCursor(
        0,
        1
      );

      lcd.print(
        "HUM:"
      );

      lcd.print(
        humidity,
        0
      );

      lcd.print(
        "%"
      );

      break;

    // --------------------------------------------------------
    // PAGE 1
    // --------------------------------------------------------

    case 1:

      lcd.setCursor(
        0,
        0
      );

      lcd.print(
        "CUR:"
      );

      lcd.print(
        current_A,
        2
      );

      lcd.print(
        "A"
      );

      lcd.setCursor(
        0,
        1
      );

      lcd.print(
        "INA:"
      );

      lcd.print(
        busVoltage,
        1
      );

      lcd.print(
        "V"
      );

      break;

    // --------------------------------------------------------
    // PAGE 2
    // --------------------------------------------------------

    case 2:

      lcd.setCursor(
        0,
        0
      );

      lcd.print(
        "VOLT:"
      );

      lcd.print(
        monitoredVoltage,
        1
      );

      lcd.print(
        "V"
      );

      lcd.setCursor(
        0,
        1
      );

      lcd.print(
        "VIB:"
      );

      lcd.print(
        vibrationValue,
        1
      );

      break;

    // --------------------------------------------------------
    // PAGE 3
    // --------------------------------------------------------

    case 3:

      lcd.setCursor(
        0,
        0
      );

      if (
        reading.fault
      )
      {
        lcd.print(
          "!!! FAULT !!!"
        );
      }
      else
      {
        lcd.print(
          "SYSTEM NORMAL"
        );
      }

      lcd.setCursor(
        0,
        1
      );

      if (
        reading.highTemperature
      )
      {
        lcd.print(
          "HIGH TEMP"
        );
      }
      else if (
        reading.overCurrent
      )
      {
        lcd.print(
          "OVER CURRENT"
        );
      }
      else if (
        reading.highVibration
        ||
        reading.vibrationDetected
      )
      {
        lcd.print(
          "HIGH VIBRATION"
        );
      }
      else if (
        reading.underVoltage
      )
      {
        lcd.print(
          "LOW VOLTAGE"
        );
      }
      else if (
        reading.overVoltage
      )
      {
        lcd.print(
          "HIGH VOLTAGE"
        );
      }
      else
      {
        lcd.print(
          "ALL OK"
        );
      }

      break;

    // --------------------------------------------------------
    // PAGE 4 - NETWORK
    // --------------------------------------------------------

    case 4:

      lcd.setCursor(
        0,
        0
      );

      if (
        WiFi.status()
        ==
        WL_CONNECTED
      )
      {
        lcd.print(
          "WIFI:CONNECTED"
        );
      }
      else
      {
        lcd.print(
          "WIFI:OFFLINE"
        );
      }

      lcd.setCursor(
        0,
        1
      );

      if (
        cloudServerOnline
      )
      {
        lcd.print(
          "CLOUD:ONLINE"
        );
      }
      else if (
        localServerOnline
      )
      {
        lcd.print(
          "LOCAL MODE"
        );
      }
      else
      {
        lcd.print(
          "BUFFER:"
        );

        lcd.print(
          bufferCount
        );
      }

      break;
  }

  lcdPage++;

  if (
    lcdPage
    >=
    LCD_PAGE_COUNT
  )
  {
    lcdPage = 0;
  }
}

// ============================================================
// MAIN LOOP
// ============================================================

void loop()
{
  unsigned long now =
    millis();

  // ----------------------------------------------------------
  // WIFI MAINTENANCE
  // ----------------------------------------------------------

  maintainWiFi();

  // ----------------------------------------------------------
  // SENSOR UPDATE
  // ----------------------------------------------------------

  if (
    now
      -
    lastSensorTime
      >=
    SENSOR_INTERVAL
  )
  {
    lastSensorTime =
      now;

    readSensors();

    SensorReading currentReading =
      createReading();

    sequenceNumber--;

    controlBuzzer(
      currentReading.fault
    );
  }

  // ----------------------------------------------------------
  // NETWORK SEND
  // ----------------------------------------------------------

  if (
    now
      -
    lastNetworkTime
      >=
    NETWORK_INTERVAL
  )
  {
    lastNetworkTime =
      now;

    sendNetworkData();
  }

  // ----------------------------------------------------------
  // LCD UPDATE
  // ----------------------------------------------------------

  if (
    now
      -
    lastLCDTime
      >=
    LCD_INTERVAL
  )
  {
    lastLCDTime =
      now;

    updateLCD();
  }

  // ----------------------------------------------------------
  // SERIAL UPDATE
  // ----------------------------------------------------------

  if (
    now
      -
    lastSerialTime
      >=
    SERIAL_INTERVAL
  )
  {
    lastSerialTime =
      now;

    printSerial();
  }

  // No long blocking delay.
  delay(10);
}
